# The environment of every FFmpeg entry, read once by the loop below: append to it before the loop
# (or later with set_property(TEST <name> APPEND PROPERTY ENVIRONMENT_MODIFICATION ...)). It does not
# hold DVV_FFMPEG_DIR: an entry that needs FFmpeg loaded adds
# "DVV_FFMPEG_DIR=set:${DVV_FFMPEG_RUNTIME_DIR}" itself, as the "available" mode does.
set(ffmpeg_test_environment ${DVV_TEST_ENVIRONMENT}
    "DVV_TEST_FFMPEG_RUNTIME_DIR=set:${DVV_FFMPEG_RUNTIME_DIR}"
    "DVV_TEST_FFMPEG_PROGRAM=set:${DVV_FFMPEG_PROGRAM}")
foreach(mode IN ITEMS available absent relative)
    if(mode STREQUAL "available")
        set(name dvv_components.ffmpeg)
        set(suite ffmpeg)
        set(environment ${ffmpeg_test_environment} "DVV_FFMPEG_DIR=set:${DVV_FFMPEG_RUNTIME_DIR}"
                        "DVV_FFMPEG_FIXTURES=set:${ffmpeg_fixtures_dir}"
                        "DVV_LAVFI_FIXTURES=set:${lavfi_fixtures_dir}")
    elseif(mode STREQUAL "absent")
        set(name dvv_components.ffmpeg_absent)
        set(suite ffmpeg-absent)
        set(environment ${ffmpeg_test_environment} "DVV_REQUIRE_FFMPEG=unset:"
                        "DVV_FFMPEG_DIR=set:${CMAKE_CURRENT_BINARY_DIR}/no-ffmpeg")
    else()
        set(name dvv_components.ffmpeg_relative)
        set(suite ffmpeg-relative)
        set(environment ${ffmpeg_test_environment} "DVV_REQUIRE_FFMPEG=unset:"
                        "DVV_FFMPEG_DIR=set:ffmpeg-relative")
    endif()
    add_test(NAME ${name} COMMAND dvv_components_ffmpeg_tests --test-suite=${suite})
    # doctest succeeds when the suite filter selects no case ("test cases: 0 | ..."); such a run
    # tests nothing and fails here. The skip path (exit 77) prints no doctest summary.
    set_tests_properties(${name} PROPERTIES
        LABELS ffmpeg
        FAIL_REGULAR_EXPRESSION "test cases: +0 \\|"
        SKIP_RETURN_CODE 77
        TIMEOUT 600
        ENVIRONMENT DVV_LEAK_CHECK=fail
        ENVIRONMENT_MODIFICATION "${environment};DVV_FFMPEG_TEST_MODE=set:${mode}")
endforeach()

# Compile-fail tests (spec §9.4). Case 0 is the control and must build; each other case must fail to
# compile with its pattern: 1 the test component, 2 FfmpegSourceProvider (Task 5a). Every target
# sees the private FFmpeg headers, so case 2 fails on the private constructor, not on an include.
set(compile_fail_source ${CMAKE_CURRENT_SOURCE_DIR}/compile_fail/direct_construction.cpp)
add_executable(dvv_components_compile_ok ${compile_fail_source})
target_compile_definitions(dvv_components_compile_ok PRIVATE DVV_COMPILE_FAIL_CASE=0)
target_include_directories(dvv_components_compile_ok PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/components)
target_link_libraries(dvv_components_compile_ok PRIVATE dvv::media dvv::ffmpeg dvv_ffmpeg_internal)
dvv_set_compiler_options(dvv_components_compile_ok)
if(MSVC)
    set(compile_fail_patterns "C2248" "C2248")
else()
    set(compile_fail_patterns "private constructor" "private constructor")
endif()
foreach(case IN ITEMS 1 2)
    set(target dvv_components_compile_fail_${case})
    add_executable(${target} EXCLUDE_FROM_ALL ${compile_fail_source})
    target_compile_definitions(${target} PRIVATE DVV_COMPILE_FAIL_CASE=${case})
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/components)
    target_link_libraries(${target} PRIVATE dvv::media dvv::ffmpeg dvv_ffmpeg_internal)
    dvv_set_compiler_options(${target})
    math(EXPR pattern_index "${case} - 1")
    list(GET compile_fail_patterns ${pattern_index} pattern)
    # cmake --build needs the top-level tree; in a dependent build (the app) dvv-components is a
    # subproject.
    add_test(NAME dvv_components.compile_fail.${case}
             COMMAND ${CMAKE_COMMAND}
                     -DBUILD_DIR=${CMAKE_BINARY_DIR}
                     -DCONFIG=$<CONFIG>
                     -DTARGET=${target}
                     "-DPATTERN=${pattern}"
                     -P ${DVV_CMAKE_DIR}/run-compile-fail.cmake)
    set_tests_properties(dvv_components.compile_fail.${case} PROPERTIES
                         RESOURCE_LOCK dvv_build_tree)
endforeach()
