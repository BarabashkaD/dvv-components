# Stub project for run-component-option-test.cmake: dvv_add_component() and
# dvv_link_enabled_components() without the dvv libraries (dvv::media and the compiler options
# are stubs).
cmake_minimum_required(VERSION 4.2)
project(dvv_component_option_test LANGUAGES CXX)
add_library(dvv::media INTERFACE IMPORTED)
function(dvv_set_compiler_options target)
endfunction()
include("${REGISTRY_MODULE}")
include("${COMPONENT_MODULE}")
if(CASE STREQUAL "old_kind")
    dvv_add_component(probe_a KIND Gadget SOURCES a.cpp)
elseif(CASE STREQUAL "bad_id")
    dvv_add_component(Probe-Bad SOURCES a.cpp)
elseif(CASE STREQUAL "twice")
    dvv_add_component(probe_a SOURCES a.cpp REGISTERS probe.same)
    dvv_add_component(probe_b SOURCES b.cpp REGISTERS probe.same)
else()
    dvv_add_component(probe_a SOURCES a.cpp REGISTERS probe.a probe.b)
    dvv_add_component(probe_b SOURCES b.cpp)
endif()
add_executable(app main.cpp)
if(CASE STREQUAL "old_link")
    dvv_components_link(app)
else()
    dvv_link_enabled_components(app)
endif()
get_property(registers GLOBAL PROPERTY DVV_COMPONENTS_REGISTERS)
get_property(ids GLOBAL PROPERTY DVV_COMPONENTS_IDS)
get_property(libraries GLOBAL PROPERTY DVV_COMPONENTS_LIBRARIES)
get_target_property(app_links app LINK_LIBRARIES)
if(TARGET dvv_component_probe_a)
    set(probe_a_target yes)
else()
    set(probe_a_target no)
endif()
message(STATUS "registers=<${registers}> ids=<${ids}> libraries=<${libraries}> probe_a_target=<${probe_a_target}> "
               "app_links=<${app_links}>")
